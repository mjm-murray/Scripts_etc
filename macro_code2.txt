Public Sub Ok_Click()
Dim Model As ModelDoc2
Dim Ok As Boolean
Dim i As Integer
Dim TopName, TopNamee, TopNameee, TopNameLaenge, VglString As String
Dim FullName, Source, Destination As String
Dim Action As String
Dim messages As String

Const swTiffScreenOrPrintCapture = 6
Const swTiffImageType = 7
Const swTiffCompressionScheme = 8
Const swTiffPrintDPI = 9
Const swTiffPrintPaperSize = 10
Const swTiffPrintScaleFactor = 11

Set swApp = CreateObject("SldWorks.Application")
Set Model = swApp.ActiveDoc
ModelType = Model.GetType
Path = JobSelect.Path
Action = ""

If Path = "" Then
    MsgBox ("Please enter a destination path")
    GoTo err1
Else
    If Dir(Path, vbDirectory) = "" Then
        MkDir Path
    End If
End If
    
If ModelType = swDocASSEMBLY Or ModelType = swDocPART Then
    If JobSelect.STEP = False And JobSelect.Para = False And JobSelect.STL = False Then
        MsgBox ("Please select an ouput type")
        GoTo err1
    End If
Else
    '2D component
    If JobSelect.PDF = False And JobSelect.DXF = False And JobSelect.DWG = False Then
        MsgBox ("Please select an ouput type")
        GoTo err1
    End If
End If
      
If ModelType = swDocASSEMBLY Then
    ReDim exportedPartNames(1)
    ReDim lightweightPartNames(1)
    ReDim suppressedPartNames(1)
    
    ' Ask if user wants to export the entire assembly as 3D
    Dim exportAsm As Integer
    exportAsm = MsgBox("Would you like to export the entire assembly as a 3D file (STEP, STL, X_T)?", vbYesNo + vbQuestion, "Export Assembly")
    
    If exportAsm = vbYes Then
        Call handle3D(Model, False) ' Export the assembly itself
    End If
    
    ' Then process individual parts
    Call traverseAssembly(Model)

ElseIf ModelType = swDocPART Then
    Call handle3D(Model, False)
Else
    Call handle2D(Model)
End If
Call cleanup(messages)

Exit Sub

err1:

End Sub

'Sub traverseAssembly(swComp As SldWorks.Component2)
Sub traverseAssembly(Model As ModelDoc2)
    'this recursively traverses all of the components in an assembly
    Dim vChildComp                  As Variant
    Dim swChildComp                 As SldWorks.Component2
    Dim i                           As Long
    Dim childlevel                  As Long
    Dim childlevel_temp             As Long
    Dim ModName                     As String
    Dim ModType As Integer
    Dim childDoc As ModelDoc2
    Dim swRootComp As SldWorks.Component2
    Dim swConf As SldWorks.Configuration
    Dim confirmResponse As Integer
    Dim suppressionState As Integer
    
    Set swConf = Model.GetActiveConfiguration
    Set swRootComp = swConf.GetRootComponent
    vChildComp = swRootComp.GetChildren

    For i = 0 To UBound(vChildComp)
        ModType = 0
        Set swChildComp = vChildComp(i)
        suppressionState = swChildComp.GetSuppression2()
        If suppressionState <> swComponentFullyResolved Then
            handleNonResolvedComp swChildComp
        Else
            Set childDoc = swChildComp.GetModelDoc2
            ModName = childDoc.GetTitle
            If Not (childDoc Is Nothing) Then
                ModType = childDoc.GetType()
            End If
            If ModType = swDocASSEMBLY Then
                'recursive
                Call traverseAssembly(childDoc)
            ElseIf ModType = swDocPART Then
                If IsInArray(ModName, exportedPartNames) Then
                    MsgBox ("skipping additional instance of " & ModName)
                Else
                    exportedPartNames(UBound(exportedPartNames)) = ModName
                    ReDim Preserve exportedPartNames(UBound(exportedPartNames) + 1)
                    confirmResponse = MsgBox("Export " + ModName + "?", vbYesNo)
                    If confirmResponse = vbYes Then
                        Call handle3D(childDoc, True)
                    End If
                End If
            Else
                MsgBox ("unexpected document type - aborting")
                Exit Sub
            End If
        End If
    Next i
End Sub


Private Sub handle3D(Model As ModelDoc2, isPartOfAssembly As Boolean)
Dim objName As String

' Get the document title
Let TopName = Model.GetTitle
Let TopNamee = Model.extension.CustomPropertyManager(confName).Get("Revision")
Let TopNameee = Model.extension.CustomPropertyManager(confName).Get("Description")
Let extensionIdx = 0
TopNameLaenge = Len(TopName)
' Get the filename without extension (ObjName)
extensionIdx = InStr(1, TopName, ".", vbTextCompare)
If extensionIdx > 1 Then
    'objName = Left(TopName, TopNameLaenge - 7)
    objName = Left(TopName, extensionIdx - 1)
Else
    objName = TopName
End If
If JobSelect.STEP Then
    ' Model in step Format abspeichern
    FullName = setupFullName(objName, TopNamee, TopNameee, "step")
    On Error GoTo err2
    'Model.SaveAs2 FullName, 0, True, True
    saveAs Model, FullName
    Action = Action & "STEP "
End If
If JobSelect.STL Then
    Dim tempPart As ModelDoc2
    ' Model in STL Format abspeichern
    FullName = setupFullName(objName, TopNamee, TopNameee, "stl")
    On Error GoTo err2
    'Model.SaveAs2 FullName, 0, True, True
    
    Rem for whatever reason, STLs require special handling when the part is in an assembly
    If isPartOfAssembly = True Then
        Call openPart(Model)
        Set tempPart = swApp.ActiveDoc
        saveAs tempPart, FullName
        Action = Action & "STL "
        Call closePart(Model.GetTitle)
    Else
        saveAs Model, FullName
    End If
End If

If JobSelect.Para Then
    FullName = setupFullName(objName, TopNamee, TopNameee, "x_t")
    On Error GoTo err2
    'Model.SaveAs2 FullName, 0, True, True
    saveAs Model, FullName
    Action = Action & "X_t "
End If

Exit Sub

err2:
MsgBox ("An error occurred in the creation, please check manually")

End Sub

Private Function IsInArray(stringToBeFound As String, arr As Variant) As Boolean
    IsInArray = (UBound(Filter(arr, stringToBeFound)) > -1)
End Function

Private Function handleNonResolvedComp(comp As SldWorks.Component2)
    Dim status As Integer
    status = comp.GetSuppression2
    If status = swComponentLightweight Or status = swComponentFullyLightweight Then
        lightweightPartNames(UBound(lightweightPartNames)) = comp.Name2
        ReDim Preserve lightweightPartNames(UBound(lightweightPartNames) + 1)
    ElseIf status <> swComponentFullyResolved Then
        suppressedPartNames(UBound(suppressedPartNames)) = comp.Name2
        ReDim Preserve suppressedPartNames(UBound(suppressedPartNames) + 1)
    End If
End Function

    Model.SaveAs2 FullName, 0, True, False
End Function

Function setupFullName(ByVal objName As Variant, ByVal TopNamee As Variant, ByVal TopNameee As Variant, ByVal extension As Variant) As String
    Dim theFullName As String
    Dim fileName As String

    ' Build filename: [PartNo]-Rev [Revision]-[Description].[extension]
    fileName = objName & "-Rev " & TopNamee & "-" & TopNameee & "." & extension

    If JobSelect.Sort Then
        If Dir(Path & "\" & UCase(extension) & "\", vbDirectory) = "" Then
            MkDir (Path & "\" & UCase(extension) & "\")
        End If
        theFullName = Path & "\" & UCase(extension) & "\" & fileName
    Else
        theFullName = Path & "\" & fileName
    End If

    setupFullName = theFullName
End Function


Function cleanup(messages As String)
JobSelect.Hide
If ModelType = swDocASSEMBLY Then
    If UBound(lightweightPartNames) > 1 Then
        messages = "The following lightweight parts were skipped:" & vbCrLf
        For i = 1 To UBound(lightweightPartNames)
            messages = messages & lightweightPartNames(i) & vbCrLf
        Next i
    End If
    If UBound(suppressedPartNames) > 1 Then
        If Len(messages) > 0 Then
            messages = messages & vbCrLf
        End If
        messages = messages & "The following suppressed parts were skipped:" & vbCrLf
        For k = 1 To UBound(suppressedPartNames)
            messages = messages + suppressedPartNames(k) & vbCrLf
        Next k
    End If
End If

MsgBox ("Completed" & vbCrLf & vbCrLf & messages)
End Function
